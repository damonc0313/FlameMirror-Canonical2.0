from typing import Optional, Union, Literal, Protocol, TypeVar
from collections.abc import Sequence

def calculate_compound_interest(
principal: Decimal,  # Financial calculation → Decimal (exact)
rate: float,         # Percentage → float acceptable
periods: int,        # Count → int (unlimited precision)
compound_frequency: Literal[“daily”, “monthly”, “yearly”] = “yearly”
) -> Decimal:
“””
Calculate compound interest with type-driven design.

```
Type Selection Decision Matrix:

1. Decimal for principal: Financial calculations require exact decimal
   representation. float introduces rounding errors (0.1 + 0.2 ≠ 0.3).
   
2. float for rate: Percentage calculations don't require exactness,
   performance benefit from float operations.
   
3. int for periods: Count values need integer semantics. Python's int
   has unlimited precision (no overflow).
   
4. Literal for frequency: Constrains inputs to valid options at type-check
   time, documents API without runtime validation overhead.

Uncertainty Weaponization: What if rate precision matters?
Constraint: If rate comes from financial data requiring exactness,
switch to Decimal. Profile to determine performance impact.

Observer Insight: Type choices visible in signature communicate
precision requirements without reading implementation.

Performance Characteristics:
Operation          | Decimal  | float | int
-------------------|----------|-------|-------
Addition           | O(n)     | O(1)  | O(n)
Multiplication     | O(n²)    | O(1)  | O(n²)
Precision          | Exact    | ~15   | Exact
                   |          | digits|

Where n = number of digits. For financial calculations with <10 digits,
Decimal overhead negligible vs correctness guarantee.
"""
getcontext().prec = 28  # Set Decimal precision

frequency_map = {"daily": 365, "monthly": 12, "yearly": 1}
n = frequency_map[compound_frequency]

# Mixed arithmetic: Decimal + float → Decimal (preserves precision)
amount = principal * (1 + Decimal(str(rate)) / n) ** (n * periods)

return amount.quantize(Decimal('0.01'))  # Round to cents
```

from collections import deque, Counter, defaultdict

class PerformanceOptimizedCache:
“””
Demonstrate data structure selection through complexity analysis.

```
Requirements:
- Fast membership testing (is item cached?)
- Fast insertion/removal
- Bounded size (LRU eviction)
- Ordered by access time

Design Question: Which data structure satisfies all requirements?

Analysis:
list:  O(n) membership, O(n) removal → unacceptable for large caches
set:   O(1) membership, but no ordering → can't implement LRU
deque: O(1) both ends, but O(n) membership → need additional structure

Solution: Hybrid approach using deque + set
"""

def __init__(self, max_size: int):
    # Optimal: deque + set combination
    self._cache_queue = deque(maxlen=max_size)  # O(1) both ends
    self._cache_set = set()  # O(1) membership
    
    """
    Dimensional Analysis of Choice:
    - 2D (Structural): Two data structures maintain consistency
    - 4D (Temporal): Deque preserves insertion order for LRU
    - Network (Relational): Set provides fast membership lookup
    - Performance: O(1) for all operations vs O(n) with list
    
    Trade-off: Memory overhead for two structures vs algorithmic
    complexity improvement. For caches >100 items, complexity
    dominates memory concerns.
    
    Apparent Contradiction: "Maintain two data structures" vs "KISS principle"
    Resolution: Algorithmic complexity trumps implementation simplicity
    when performance requirements demand it. Document trade-off explicitly.
    """

def add(self, item: str) -> None:
    """
    Add item to cache with O(1) complexity.
    
    Observer Insight: maxlen parameter provides automatic eviction,
    eliminating manual size management and potential off-by-one bugs.
    """
    if item not in self._cache_set:
        if len(self._cache_queue) >= self._cache_queue.maxlen:
            evicted = self._cache_queue[0]
            self._cache_set.discard(evicted)
        
        self._cache_queue.append(item)
        self._cache_set.add(item)

def contains(self, item: str) -> bool:
    """O(1) membership test via set."""
    return item in self._cache_set
```

import logging

def calculate_risk_adjusted_return(

*,  # Keyword-only marker

annualization_factor: float = 252.0
) -> float:
“””
Calculate Sharpe ratio with parameter design demonstrating PEP 570.

```
Args:
    returns: Historical return series (positional-only for API stability)
    risk_free_rate: Risk-free rate for excess return (positional-only)
    window_size: Lookback period (keyword-only for clarity)
    annualization_factor: Scaling factor for annual return (keyword-only)

Returns:
    Annualized Sharpe ratio

Raises:
    ValueError: If returns empty or window_size exceeds data length

Design Rationale:
1. Positional-only (/) for required parameters:
   - Prevents breaking changes if parameter names refactored
   - Clear that order matters semantically
   - Common in financial calculations (return, rate ordering standard)

2. Keyword-only (*) for optional parameters:
   - Explicit parameter names prevent confusion
   - Self-documenting call sites
   - Prevents positional argument mistakes with multiple numeric params

Temporal Perspective:
- Pre-Python 3.8: All parameters could be passed positionally or by keyword
- Python 3.8+ (PEP 570): Explicit control over parameter passing style
- Future: Type systems may encode calling conventions statically

Observer Insight: Parameter markers encode API stability guarantees visible
in signature without documentation reading.

Parameter Decision Matrix:
Use / (positional-only) when:
- Parameter order has semantic meaning
- Names may change in future versions
- Common pattern (x, y coordinates)

Use * (keyword-only) when:
- >3 parameters of same type
- Parameters have similar types (multiple floats)
- Optional parameters with defaults

Use neither when:
- Single parameter
- Parameters have distinct types
- Backward compatibility required
"""
if not returns:
    raise ValueError("Cannot calculate Sharpe ratio on empty return series")

if len(returns) < window_size:
    raise ValueError(
        f"Insufficient data: {len(returns)} returns for window size {window_size}"
    )

excess_returns = [r - risk_free_rate for r in returns[-window_size:]]
mean_excess = sum(excess_returns) / len(excess_returns)

if len(excess_returns) < 2:
    return 0.0

variance = sum((r - mean_excess) ** 2 for r in excess_returns) / (len(excess_returns) - 1)
std_dev = variance ** 0.5

if std_dev == 0:
    return 0.0

return (mean_excess / std_dev) * (annualization_factor ** 0.5)
```

def timing_decorator(func):
“””
Demonstrate decorator pattern for cross-cutting concerns.

```
Design Question: When to use decorators vs explicit logging?

Decorator advantages:
- Separates timing concern from business logic
- Reusable across functions
- Maintains clean function signatures

Trade-offs:
- Adds indirection (harder to trace for beginners)

def wrapper(*args, **kwargs):
    import time
    start = time.perf_counter()
    
    try:
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        logging.info(f"{func.__name__} completed in {elapsed:.4f}s")
        return result
    except Exception as e:
        elapsed = time.perf_counter() - start
        logging.error(f"{func.__name__} failed after {elapsed:.4f}s: {e}")
        raise

return wrapper
```

# Try-Except-Else-Finally Pattern

def process_configuration(filename: str) -> dict:
“””
Demonstrate exception handling as architectural control flow.

```
Question: Why separate try/except/else instead of single try block?
Answer: Distinguishes error-prone operations from success-path logic.

Apparent Contradiction: "Keep try blocks minimal" vs "Related operations together"
Resolution: Try block contains operations sharing failure modes. Success-path
logic in else clause prevents false attribution of exceptions.

Pattern Analysis:
try:    Operations that may fail
except: Recovery strategies for specific failures  
else:   Success-path code (only if no exception)
finally: Cleanup (always executes)
"""
try:
    with open(filename, 'r') as f:
        content = f.read()
except FileNotFoundError:
    logging.warning(f"Config file {filename} not found, using defaults")
    return get_default_config()
except PermissionError as e:
    logging.error(f"Permission denied: {filename}")
    raise ConfigurationError(f"Cannot access {filename}") from e
else:
    # Success path - only executes if no exception
    # If parse_config() raises, we know it's parsing failure, not I/O
    logging.info(f"Successfully loaded configuration from {filename}")
    return parse_config(content)
```

# Context Managers

@contextmanager
def temporary_directory(prefix: str = “tmp”) -> Generator[Path, None, None]:
“””
Provide temporary directory with guaranteed cleanup.

```
Meta-Pattern Recognition: What makes this better than manual cleanup?
1. Impossible to forget cleanup (language enforces it)
2. Cleanup happens even if exception occurs
3. Clear resource lifecycle in calling code

Trade-off Analysis:
- Pro: Safety, clarity, composability
- Con: Slight indirection vs explicit try-finally
- Decision: Use for any resource requiring cleanup

Uncertainty Navigation: Should cleanup suppress exceptions?
Answer: No - if cleanup fails, that's also error condition.
But primary exception takes precedence (Python default behavior).

Evolution Recognition:
Historical: Manual try-finally patterns
Modern (PEP 343): Context managers abstract cleanup
Future: Async context managers for async resources
"""
temp_path = None
try:
    temp_path = Path(tempfile.mkdtemp(prefix=prefix))
    logging.debug(f"Created temporary directory: {temp_path}")
    yield temp_path
except Exception as e:
    logging.error(f"Error in temporary directory context: {e}")
    raise
finally:
    if temp_path and temp_path.exists():
        try:
            shutil.rmtree(temp_path)
            logging.debug(f"Cleaned up: {temp_path}")
        except Exception as cleanup_error:
            # Cleanup failure shouldn't mask primary exception
            logging.warning(f"Failed to cleanup {temp_path}: {cleanup_error}")
```

# Logging Integration

def critical_data_pipeline(data: list[dict]) -> ‘ProcessResult’:
“””
Demonstrate logging and exception handling as complementary strategies.

```
Design Question: When to log, when to raise, when both?
Decision Matrix:
    IF (caller_should_handle) THEN raise
    IF (diagnostic_context_needed) THEN log_before_raise
    IF (recoverable_degradation) THEN log_and_continue
    IF (unexpected_but_continuing) THEN log_warning

Meta-question: Does every exception need logging?
Answer: No - exceptions raised through multiple layers create log spam.
Log at boundary where handling decision made.

Logging Level Decision Tree:
DEBUG:   Entry/exit, variable states (verbose, disabled in production)
INFO:    Normal operations, successful completions
WARNING: Recoverable errors, degraded functionality
ERROR:   Failures requiring attention, cannot complete operation
CRITICAL: System-level failures, service unavailability
"""
processed = []
failures = []

logging.info(
    f"Starting pipeline with {len(data)} items",
    extra={'data_hash': hash(str(data))}
)

for item in data:
    try:
        result = validate_and_process(item)
        processed.append(result)
        
    except ValidationError as e:
        # Recoverable - log and continue with degraded result
        logging.warning(
            f"Validation failed for item {item.get('id', 'unknown')}",
            extra={'item': item, 'error': str(e)},
            exc_info=False  # No traceback for expected errors
        )
        failures.append({'item': item, 'error': str(e)})
        
    except Exception as e:
        # Unexpected - log with full context and raise
        logging.exception(
            f"Unexpected error processing item {item.get('id', 'unknown')}",
            extra={'item': item}
        )
        raise PipelineError(f"Pipeline failed at item {item}") from e

logging.info(
    f"Pipeline completed: {len(processed)} succeeded, {len(failures)} failed",
    extra={
        'success_rate': len(processed) / len(data) if data else 0,
        'failures': failures[:5]
    }
)

return ProcessResult(processed=processed, failures=failures)
```

# Assertions vs Validation

```
CORRECT - Explicit validation for production:
```

# Magic Methods

@dataclass
class Price:
“””
Demonstrate magic method implementation for custom operators.

```
Design Question: When to implement operator overloading?

Appropriate for:
- Mathematical types (Vector, Matrix, Complex)
- Units with natural operations (Money, Distance, Duration)
- Collections with mathematical semantics

Avoid for:
- Business logic classes (Account, Order) where operators unclear
- Side-effect operations (+ shouldn't have side effects)

Meta-Pattern Recognition: Magic methods enable domain-specific languages
within Python. Price arithmetic reads naturally: price1 + price2
"""
amount: float
currency: str

def __add__(self, other: 'Price') -> 'Price':
    """Implement addition for Price objects."""
    if self.currency != other.currency:
        raise ValueError(f"Cannot add {self.currency} and {other.currency}")
    return Price(self.amount + other.amount, self.currency)

def __mul__(self, scalar: float) -> 'Price':
    """Implement scalar multiplication."""
    return Price(self.amount * scalar, self.currency)

def __eq__(self, other: object) -> bool:
    """Implement equality comparison."""
    if not isinstance(other, Price):
        return NotImplemented
    return self.amount == other.amount and self.currency == other.currency

def __lt__(self, other: 'Price') -> bool:
    """
    Implement less-than for sorting.
    
    Meta-Pattern Recognition: Implementing __lt__ enables:
    - sorted(prices)
    - max(prices)
    - min(prices)
    
    With functools.total_ordering, implementing __eq__ and one
    comparison generates the rest.
    """
    if self.currency != other.currency:
        raise ValueError(f"Cannot compare {self.currency} and {other.currency}")
    return self.amount < other.amount

def __str__(self) -> str:
    """Human-readable representation for display."""
    return f"{self.amount:.2f} {self.currency}"

def __repr__(self) -> str:
    """
    Unambiguous representation for debugging.
    
    Design Goal: repr output should enable object recreation:
    eval(repr(obj)) == obj
    
    Observer Insight: __repr__ for developers (debugging),
    __str__ for end users (display).
    """
    return f"Price(amount={self.amount!r}, currency={self.currency!r})"
```
